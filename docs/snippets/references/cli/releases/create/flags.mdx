<ParamField path="--schemas" type="string">
  A directory containing the config schemas to include in the release. May be specified multiple times for multiple directories.
  
  Must be specified if no schema files are provided.

  Examples: `./schemas`
</ParamField>

<ParamField path="--schema" type="string">
  The path to a config schema to include in the release. May be specified multiple times to include multiple schemas. 

  Must be specified if no schema directory is provided.

  Examples: `./schemas/schema1.yaml`, `./schemas/schema2.yaml`
</ParamField>

<ParamField path="--file-rules" type="string">
  A directory containing [file rule](/data-uploads/primitives/file-rules#file-format) YAML files to include in the release. May be specified multiple times for multiple directories. Only YAML files in the directory are picked up.

  Examples: `./file-rules/`
</ParamField>

<ParamField path="--file-rule" type="string">
  The path to a [file rule](/data-uploads/primitives/file-rules#file-format) YAML file to include in the release. May be specified multiple times to include multiple rules.

  Examples: `./file-rules/robot-logs.yaml`
</ParamField>
